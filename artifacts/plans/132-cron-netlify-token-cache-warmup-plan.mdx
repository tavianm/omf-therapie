```ts
async function warmAvailabilityCache(): Promise<void> {
  const modes = ['in-person', 'video'] as const;
  const durations = [60, 90] as const;
  const now = new Date();
  const end = new Date(now.getTime() + 4 * 7 * 24 * 3600 * 1000);
  await Promise.allSettled(
    modes.flatMap(m => durations.map(async d => {
      const slots = await getAvailableSlots(now, end, d, m, []);
      const key = buildAvailabilityCacheKey(m, d, 4, now);
      await setCachedAvailability(key, slots, 900);
    })),
  ); // + log résumé, captureAndFlush sur rejets
}
```
